```python
import matplotlib.pyplot as plt

class Member:
    def __init__(self, name: str, poems: int = 0) -> None:
        self.name = name
        self.poems = poems

    def poem(self) -> str:
        return f"{self.name} wrote {self.poems} poems"


for girl in [Member("Sayori", 12), Member("Yuri", 15)]:
    print(girl.poem())
```
